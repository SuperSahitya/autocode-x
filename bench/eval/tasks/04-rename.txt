-- check.test.mjs --
import assert from "node:assert/strict";
import { readFileSync, readdirSync } from "node:fs";
import { join } from "node:path";
import { test } from "node:test";
import { computeTotal } from "./src/cart.js";
import { calcTotals } from "./src/calcTotals.js";
import { discounted } from "./src/checkout/discount.js";
import { summary } from "./src/checkout/summary.js";
import { invoiceAmount } from "./src/invoice.js";

const lines = [{ price: 100, qty: 2 }, { price: 50, qty: 1 }];
const walk = (dir) => readdirSync(dir, { withFileTypes: true }).flatMap((e) => (e.isDirectory() ? walk(join(dir, e.name)) : [join(dir, e.name)]));

test("every caller works under the new name", () => {
  assert.equal(computeTotal(lines), 250);
  assert.equal(invoiceAmount({ lines, shipping: 10 }), 260);
  assert.equal(summary(lines), "Total: 250");
  assert.equal(discounted(lines, 10), 225);
});
test("the old name is gone, and the similar one survives", () => {
  for (const file of walk("src")) assert.doesNotMatch(readFileSync(file, "utf8"), /calcTotal(?!s)/, file);
  assert.deepEqual(calcTotals([[1, 2]]), [2]);
});
