import { pageCount, paginate } from "./pagination.js";
import { allProducts } from "./products.js";

export function listing(page, size = 2) {
  const products = allProducts();
  return { page, pages: pageCount(products.length, size), items: paginate(products, page, size) };
}

import assert from "node:assert/strict";
import { test } from "node:test";
import { listing } from "./src/listing.js";
import { pageCount, paginate } from "./src/pagination.js";

test("first page starts at the first item", () => {
  assert.deepEqual(paginate([1, 2, 3, 4, 5], 1, 2), [1, 2]);
});
test("last partial page is reachable", () => {
  assert.equal(pageCount(5, 2), 3);
  assert.equal(pageCount(4, 2), 2);
  assert.deepEqual(paginate([1, 2, 3, 4, 5], 3, 2), [5]);
});
test("listing reports both", () => {
  assert.deepEqual(listing(3), { page: 3, pages: 3, items: ["rug"] });
