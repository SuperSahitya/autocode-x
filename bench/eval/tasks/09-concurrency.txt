kind: bug
runAll in src/pool.js ignores its limit and starts everything at once, which is getting us rate limited. It should never have more than limit tasks running, return results in the same order as the input, and reject if any task rejects.
-- repo/package.json --
{ "type": "module", "scripts": { "test": "node --test" } }
-- repo/src/pool.js --
/** Runs async functions with at most `limit` going at once. */
export async function runAll(tasks, limit) {
  const results = [];
  let running = 0;
  for (const task of tasks) {
    running++;
    task().then((value) => {
      results.push(value);
      running--;
    });
  }
  while (running > 0) await new Promise((r) => setTimeout(r, 1));
  return results;
}

-- solution/src/pool.js --
/** Runs async functions with at most `limit` going at once. */
export async function runAll(tasks, limit) {
  const results = new Array(tasks.length);
  let next = 0;
  const worker = async () => {
    while (next < tasks.length) {
      const i = next++;
      results[i] = await tasks[i]();
    }
  };
  await Promise.all(Array.from({ length: Math.min(limit, tasks.length) }, worker));
  return results;
}
-- check.test.mjs --
import assert from "node:assert/strict";
import { test } from "node:test";
import { fetchAll } from "./src/fetchAll.js";
import { runAll } from "./src/pool.js";

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

test("never exceeds the limit, and uses it", async () => {
  let running = 0;
  let peak = 0;
  const tasks = Array.from({ length: 12 }, () => async () => {
    peak = Math.max(peak, ++running);
    await sleep(5);
    running--;
  });
  await runAll(tasks, 3);
  assert.equal(peak, 3);
});
test("results match the input order", async () => {
  const delays = [30, 1, 15, 5];
  const results = await runAll(delays.map((d) => async () => (await sleep(d), d)), 2);
  assert.deepEqual(results, delays);
});
test("a rejection rejects the whole run", async () => {
  await assert.rejects(runAll([async () => 1, async () => { throw new Error("boom"); }, async () => 3], 2), /boom/);
});
test("handles no tasks, and a limit above the task count", async () => {
  assert.deepEqual(await runAll([], 3), []);
  assert.deepEqual(await runAll([async () => 1], 5), [1]);
});
test("callers work", async () => {
  assert.deepEqual(await fetchAll(["a", "b"], async (u) => u + "!"), ["a!", "b!"]);
});
